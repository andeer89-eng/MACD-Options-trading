    prevLow        := curLow

    curLow         := low

    curLow         := low

    curHigh := math.max(high, curHigh)

type LineSeg
    line  active    = na
    float activeVal = na
    array<line> hist = na

type ZoneSeg
    box   active    = na
    float activeTop = na
    float activeBot = na
    array<box> hist = na

// Helper: Clear line history efficiently
f_clearHistory(array<line> histArr) =>
    if not na(histArr) and array.size(histArr) > 0
        for i = 0 to array.size(histArr) - 1
            line.delete(array.get(histArr, i))
        array.clear(histArr)

// Helper: Clear box history efficiently
f_clearHistoryBoxes(array<box> histArr) =>
    if not na(histArr) and array.size(histArr) > 0
        for i = 0 to array.size(histArr) - 1
            box.delete(array.get(histArr, i))
        array.clear(histArr)

// Update line segments - procedural function with explicit typed return for v6
f_updateLine(LineSeg s, bool show, float val, color col, int width, bool reset) =>
    if na(s.hist)
        s.hist := array.new<line>()
    
    if reset
        line.delete(s.active)
        s.active := na
        s.activeVal := na
        f_clearHistory(s.hist)
    
    if not show or na(val)
        line.delete(s.active)
        s.active := na
        s.activeVal := na
        f_clearHistory(s.hist)
    else
        if na(s.active)
            s.active    := line.new(time, val, time, val, xloc = xloc.bar_time, color = col, width = width)
            s.activeVal := val
        else if val != s.activeVal
            line.set_xy2(s.active, time, s.activeVal)
            array.push(s.hist, s.active)
            s.active    := line.new(time, val, time, val, xloc = xloc.bar_time, color = col, width = width)
            s.activeVal := val
        else
            line.set_xy2(s.active, time, val)
    float(na) // Explicit return type to satisfy v6 compiler

// Update zone boxes - procedural function with explicit typed return for v6
f_updateZone(ZoneSeg z, bool show, float top, float bot, color bg, bool reset) =>
    if na(z.hist)
        z.hist := array.new<box>()
    
    if reset
        box.delete(z.active)
        z.active := na
        z.activeTop := na
        z.activeBot := na
        f_clearHistoryBoxes(z.hist)
    
    if not show or na(top) or na(bot)
        box.delete(z.active)
        z.active := na
        z.activeTop := na
        z.activeBot := na
        f_clearHistoryBoxes(z.hist)
    else
        float tHi = math.max(top, bot)
        float tLo = math.min(top, bot)
        if na(z.active)
            z.active    := box.new(time, tHi, time, tLo, xloc = xloc.bar_time, bgcolor = bg, border_color = na)
            z.activeTop := top
            z.activeBot := bot
        else if top != z.activeTop or bot != z.activeBot
            box.set_right(z.active, time)
            array.push(z.hist, z.active)
            z.active    := box.new(time, tHi, time, tLo, xloc = xloc.bar_time, bgcolor = bg, border_color = na)
            z.activeTop := top
            z.activeBot := bot
        else
            box.set_right(z.active, time)
    float(na) // Explicit return type to satisfy v6 compiler

// RECLAIM SIGNALS

// State machine:
// BULL: ARM at 78.6% → FIRE when crossing above 61.8% → DISARM
// BEAR: ARM at 78.6% → FIRE when crossing below 61.8% → DISARM
// Shallow variant uses 61.8%→50% levels

bool useDeep    = reclaimType == "Deep (786→618)" or reclaimType == "Either"
bool useShallow = reclaimType == "Shallow (618→50)" or reclaimType == "Either"

var bool bullArmedDeep    = false
var bool bullArmedShallow = false
var bool bearArmedDeep    = false
var bool bearArmedShallow = false

if isNewPeriod
    bullArmedDeep    := false
    bullArmedShallow := false
    bearArmedDeep    := false
    bearArmedShallow := false

if validRange
    if close <= level786Bull
        bullArmedDeep    := true
    if close <= level618Bull
        bullArmedShallow := true
    if close >= level786Bear
        bearArmedDeep    := true
    if close >= level618Bear
        bearArmedShallow := true

// Crossover detection
bool xUpBull618 = ta.crossover (close, level618Bull)
bool xUpBull50  = ta.crossover (close, level50)
bool xDnBear618 = ta.crossunder(close, level618Bear)
bool xDnBear50  = ta.crossunder(close, level50)
bool xUpExtUp   = ta.crossover (close, level1272Up)
bool xDnExtDn   = ta.crossunder(close, level1272Dn)

// Reclaim signal conditions
bool bullReclaimDeep    = bullArmedDeep    and xUpBull618 and validRange
bool bullReclaimShallow = bullArmedShallow and xUpBull50  and validRange
bool bearReclaimDeep    = bearArmedDeep    and xDnBear618 and validRange
bool bearReclaimShallow = bearArmedShallow and xDnBear50  and validRange

// Disarm after firing to prevent duplicate signals (fixed "then" syntax)
if bullReclaimDeep
    bullArmedDeep := false
if bullReclaimShallow
    bullArmedShallow := false
if bearReclaimDeep
    bearArmedDeep := false
if bearReclaimShallow
    bearArmedShallow := false

bool bullSignal = (useDeep and bullReclaimDeep) or (useShallow and bullReclaimShallow)
bool bearSignal = (useDeep and bearReclaimDeep) or (useShallow and bearReclaimShallow)
