// §2  HELPER FUNCTIONS

f_safeDiv(n, d, fallback) => d != 0 and not na(d) and not na(n) ? n / d : fallback

f_safeLogRet(price, prevPrice) =>
    prevPrice > 0 and price > 0 and not na(price) and not na(prevPrice) ? math.log(price / prevPrice) : na

f_ma(src, len, t) =>
    switch t
        "EMA"  => ta.ema(src, len)
        "WMA"  => ta.wma(src, len)
        "VWMA" => ta.vwma(src, len)
        "HMA"  => ta.hma(src, math.max(len, 2))
        =>        ta.sma(src, len)

f_wMA_roc(malen, matype, slen) =>
    ma   = f_ma(close, malen, matype)
    prev = ma[slen]
    roc  = not na(prev) and prev > 0.0 ? (ma - prev) / prev * 100.0 : 0.0
    [ma, roc]

f_wRVOL(len) =>
    avg = ta.sma(volume, len)
    avg > 0.0 ? volume / avg : na

f_swingLow(len) => ta.lowest(low, len)

f_compression(len) =>
    hi  = ta.highest(high, len)
    lo  = ta.lowest(low, len)
    rng = hi - lo
    avg = ta.sma(rng, len)
    avg > 0 ? rng / avg : na

f_wATRContext(atrlen, plen) =>
    r   = ta.atr(atrlen) / close
    p25 = ta.percentile_linear_interpolation(r, plen, 25)
    [r, p25]

f_wOrderFlow(bodyLen) =>
    barRng   = math.max(high - low, syminfo.mintick)
    bv       = volume * (close - low)  / barRng
    sv       = volume * (high - close) / barRng
    body     = math.abs(close - open)
    bodyAvg  = ta.sma(body, bodyLen)
    volAvg   = ta.sma(volume, bodyLen)
    absorb   = volume > volAvg * 1.8 and body < bodyAvg * 0.5
    hAccum   = bv > sv and close <= close[1] and volume > volAvg
    hDistrib = sv > bv and close >= close[1] and volume > volAvg
    [bv, sv, absorb, hAccum, hDistrib]

f_volRegime(atrRatio, lookback) =>
    p25 = ta.percentile_linear_interpolation(atrRatio, lookback, 25)
    p75 = ta.percentile_linear_interpolation(atrRatio, lookback, 75)
    atrRatio < p25 ? "LOW" : atrRatio > p75 ? "HIGH" : "MED"

f_frictionPenalty(rocPct, totalFrictionBps) =>
    frictionDecimal = totalFrictionBps / 10000.0
    minEdgeRequired = frictionDecimal * 3.0
    math.abs(rocPct) / 100.0 >= minEdgeRequired ? 0 : -1

buildJsonAlert(signalType, ticker, stageVal, convictionVal, stopLevel, audit) =>
    json = '{"signal":"' + signalType + '",'

    json := json + '"audit":"' + audit + '",'
    json := json + '"env":"' + deployEnv + '",'
    json := json + '"version":"3.3.0-inst"}'
    json

// ════════════════════════════════════════════════════════════════════════
// §4  DERIVED VALUES

wValid   = not na(wMA) and wMA > 0.0
wRising  = wRoc > slopeMin
wFalling = wRoc < -slopeMin
wFlat    = not wRising and not wFalling
wBodLo   = math.min(wOpen, wClose)
wBodHi   = math.max(wOpen, wClose)

volRegimeStr = useAdaptive ? f_volRegime(wAtrRatio, volLookback) : "MED"
adaptiveFactor = useAdaptive ? (volRegimeStr == "HIGH" ? 1.0 + adaptiveSensitivity * 0.5 : volRegimeStr == "LOW" ? 1.0 - adaptiveSensitivity * 0.3 : 1.0) : 1.0
adaptiveBufPct = bufPct * adaptiveFactor
wBuf     = wValid ? wMA * (adaptiveBufPct / 100.0) : 0.0

isNewWk  = ta.change(wBarIdx) != 0
nearMA   = wValid ? math.abs(wClose - wMA) <= wBuf : false

rvol     = not na(wRVOL) ? wRVOL : 1.0
volScore = not useVol ? 1 : rvol >= rvolMin * 1.5 ? 2 : rvol >= rvolMin ? 1 : 0
volOK    = volScore > 0
volDryUp = rvol < 0.8

wOFRatio = not na(wBuyVol) and not na(wSellVol) and wSellVol > 0 ? f_safeDiv(wBuyVol, wSellVol, 1.0) : 1.0

rsShortOK = useRS and wBenchValid and not na(wClose) and not na(wClose[13]) and wClose[13] > 0
rsLongOK  = useRS and wBenchValid and not na(wClose) and not na(wClose[rsLen]) and wClose[rsLen] > 0
rsShort   = rsShortOK ? f_safeLogRet(wClose, wClose[13]) - f_safeLogRet(wBench, wBench[13]) : na
rsLong    = rsLongOK  ? f_safeLogRet(wClose, wClose[rsLen]) - f_safeLogRet(wBench, wBench[rsLen]) : na
rsVal     = not na(rsShort) and not na(rsLong) ? rsShort * 0.6 + rsLong * 0.4 : not na(rsLong) ? rsLong : rsShort
rsUp      = not na(rsVal) and rsVal > 0.0

rsSma4    = ta.sma(rsVal, 4)
rsSma12   = ta.sma(rsVal, 12)
rsTrend   = not na(rsSma4) and not na(rsSma12) and rsSma4 > rsSma12
rsSma4Roc = ta.roc(rsSma4, 4)
rsAccel   = not na(rsSma4) and not na(rsSma4[4]) and rsSma4Roc > 0

marketBull = useRS and not na(wBench40) and not na(wBench40p4) and wBench40 > wBench40p4
wCompAvg = ta.sma(wCompRatio, 20)

wHighest5 = ta.highest(wHigh, 5)
wLowest5  = ta.lowest(wLow,  5)
wHH = not na(wHigh) and not na(wHighest5[1]) and wHigh > wHighest5[1]
wHL = not na(wLow)  and not na(wLowest5[1])  and wLow  > wLowest5[1]
isStructureUp = wHH and wHL

totalFrictionBps = spreadBps + commissionBps + slippageBps
frictionPenalty = modelFriction ? f_frictionPenalty(wRoc, totalFrictionBps) : 0

safetyMultiplier = (deployEnv == "live" and safetyMode) ? 1.5 : 1.0
adaptiveConfirmRaw = useAdaptive ? (confirmBars * (volRegimeStr == "HIGH" ? 1.3 : volRegimeStr == "LOW" ? 0.8 : 1.0) * safetyMultiplier) : confirmBars
adaptiveConfirm = math.max(1, math.min(5, math.round(adaptiveConfirmRaw)))


// ════════════════════════════════════════════════════════════════════════
// §5  STAGE DETERMINATION
// ════════════════════════════════════════════════════════════════════════

isS2raw = wValid and wBodLo > wMA + wBuf and wRising and isStructureUp

var int  s2streak = 0
var int  s4streak = 0

if not skInit and wValid
    s2streak := isS2raw ? 1 : 0
    s4streak := isS4raw ? 1 : 0
    skInit   := true
else if isNewWk
    s2streak := isS2raw ? s2streak + 1 : 0
    s4streak := isS4raw ? s4streak + 1 : 0

isS2conf = isS2raw and s2streak >= adaptiveConfirm and mtfConfirm
isS4conf = isS4raw and s4streak >= adaptiveConfirm and mtfConfirm

weakMomentum = wRoc < slopeMin * 0.5

var int trend = 0
int stage     = 0

if wValid
    if isS2conf
        trend := 1
        stage := 2
    else if isS4conf
        trend := -1
        stage := 4
    else if trend == 1 and (wFlat or nearMA) and weakMomentum
        stage := 3
    else if trend == -1 and (wFlat or nearMA)
        stage := 1
    else if trend == 1
        stage := 2
    else if trend == -1
        stage := 1

s2Pullback = stage == 2 and not isS2conf
s3Quality = stage == 3 and (wFlat or nearMA) and weakMomentum
s1B = stage == 1 and not na(wClose) and not na(wMA) and wClose > wMA + wBuf * 0.5 and wRising

rangeExpanding = not na(wCompRatio) and not na(wCompAvg) and wCompRatio > wCompAvg
earlyS2 = wValid and stage != 2 and wClose > wMA and wRising and wRoc > slopeMin * 1.5 and wClose > wMA + wBuf * 0.25 and wBodLo <= wMA + wBuf and rangeExpanding

failedS2early = stage == 2 and not na(wClose) and not na(wMA) and wClose < wMA and wRoc < slopeMin

        dur     := 1

    else if isNewWk

s2Late  = stage == 2 and not s2Pullback and dur > 40 and not wRising
s2Decay = stage == 2 and not s2Pullback and dur > 30 ? math.min(1.0, float(dur - 30) / 20.0) : 0.0
s2Alpha = int(math.round(s2Decay * 35.0))

volContract = (stage == 1 or stage == 3) and not na(wAtrRatio) and not na(wAtrRatioP25) and wAtrRatio < wAtrRatioP25
tightRange  = stage == 1 and not na(wCompRatio) and not na(wCompAvg) and wCompRatio < wCompAvg * 0.8
coiledBase = stage == 1 and volContract and tightRange and volDryUp

int bqVol   = volScore >= 1 ? 1 : 0
int bqRange = tightRange    ? 1 : 0
int bqSlope = wRising       ? 1 : 0
int breakoutQuality = bqVol + bqRange + bqSlope

var bool p1B            = false
var bool pEarlyS2       = false
var bool pFailedS2      = false
var bool pFailedS2early = false

s1Emg         = stgCh and stage == 1
s2Emg         = stgCh and stage == 2
s3Emg         = stgCh and stage == 3
s4Emg         = stgCh and stage == 4

earlyS2Emg    = earlyS2 and not pEarlyS2
failedS2Emg   = failedS2 and not pFailedS2

pFailedS2      := failedS2

// §8  CONVICTION SCORE
// ════════════════════════════════════════════════════════════════════════

volBreakout = isS2raw and rvol >= rvolMin

int convBase  = (stage == 2 or stage == 4) ? (useRS and not marketBull and stage == 2 ? 0 : 1) : 0
int convVol   = convBase > 0 and volBreakout ? volScore : 0
int convSlope = convBase > 0 ? (math.abs(wRoc) > slopeMin * 3 ? 2 : math.abs(wRoc) > slopeMin * 1.5 ? 1 : 0) : 0

int convRST   = convBase > 0 and useRS and rsTrend and rsAccel and stage == 2 ? 1 : 0
int convOF    = convBase > 0 and stage == 2 and not na(wOFRatio) and wOFRatio > 1.2 ? 1 : 0

int convFriction = modelFriction ? frictionPenalty : 0
correlationPenalty = maxCorrelation < 0.5 ? 0 : -int(math.round((maxCorrelation - 0.5) * 2.0))

int convRaw   = convBase + convVol + convSlope + convRS + convRST + convOF + convFriction + correlationPenalty
int conv      = math.max(0, math.min(5, convRaw))


// ════════════════════════════════════════════════════════════════════════
// §9  POSITION SIZING MODULE
// ════════════════════════════════════════════════════════════════════════

stopDistance = showStop and stage == 2 and not s2Pullback and not na(wATR) ? wATR * atrMult : wClose * 0.02
basePositionSize = stopDistance > 0 ? f_safeDiv(riskAmount, stopDistance, 0) : 0

volatilityFactor = volatilityAdj ? f_safeDiv(2.0, atrPct / 2.0, 1.0) : 1.0
volAdjustedSize = basePositionSize * volatilityFactor

corrFactor = maxCorrelation < 0.5 ? 1.0 : 1.0 - (maxCorrelation - 0.5) * 2.0
corrAdjustedSize = volAdjustedSize * corrFactor

exposureCap = accountSize * portfolioExposure / 100 / wClose
finalPositionSize = math.min(corrAdjustedSize, exposureCap)

positionSize = deployEnv == "live" and safetyMode ? finalPositionSize * 0.75 : finalPositionSize

        maxDrawdown := math.max(maxDrawdown, drawdown)
        cumulativeHypotheticalPnL := currentEquity
