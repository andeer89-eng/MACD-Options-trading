// Processes pending pivot levels into active levels after confirmation
processPending(array<float> pendL, array<int> pendB, array<float> active, bool isRes) =>
    array<float> newPendL = array.new<float>()
    array<int>   newPendB = array.new<int>()
    
    for i = 0 to array.size(pendL) - 1
        float lvl = array.get(pendL, i)
        int   det = array.get(pendB, i)
        bool  violated = isRes ? close > lvl : close < lvl
        
        if not violated
            if bar_index - det >= confirmBars
                // Shift active levels to make room for newest
                for j = 0 to array.size(active) - 2
                    array.set(active, j, array.get(active, j + 1))
                array.set(active, array.size(active) - 1, lvl)
            else
                array.push(newPendL, lvl)
                array.push(newPendB, det)
    
    array.clear(pendL)
    array.clear(pendB)
    for i = 0 to array.size(newPendL) - 1
        array.push(pendL, array.get(newPendL, i))
        array.push(pendB, array.get(newPendB, i))

// Clears active levels if they are breached by price
checkBreaks(array<float> active, bool isRes) =>
    for i = 0 to array.size(active) - 1
        float lvl = array.get(active, i)
        if not na(lvl)
            bool broken = isRes ? close > lvl : close < lvl
            if broken
                array.set(active, i, na)

// --- Stratification Fills (Boxes) ---
var array<box> voilaBoxes = array.new<box>()

    for b in voilaBoxes
        box.delete(b)
    array.clear(voilaBoxes)
    
    array<float> allLvls = array.new<float>()
    for i = 0 to 2
        if not na(array.get(aDR, i))
            array.push(allLvls, array.get(aDR, i))
        if not na(array.get(aDS, i))
            array.push(allLvls, array.get(aDS, i))
        if not na(array.get(aWR, i))
            array.push(allLvls, array.get(aWR, i))
        if not na(array.get(aWS, i))
            array.push(allLvls, array.get(aWS, i))
    
    if array.size(allLvls) >= 2
        array.sort(allLvls, order.ascending)
        
        // Filter unique levels with some tolerance
        array<float> uniqueLvls = array.new<float>()
        float prevLvl = na
        for lvl in allLvls
            if na(prevLvl) or math.abs(lvl - prevLvl) > syminfo.mintick * 10
                array.push(uniqueLvls, lvl)
                prevLvl := lvl
        
        if array.size(uniqueLvls) >= 2
            for i = 0 to array.size(uniqueLvls) - 2
                float bot = array.get(uniqueLvls, i)
                float top = array.get(uniqueLvls, i + 1)
                int alpha = 90 - (i * 5) % 25
                box b = box.new(bar_index, top, bar_index + 1, bot, border_color=na, bgcolor=color.new(C_SILVER, alpha), extend=extend.right)
                array.push(voilaBoxes, b)
